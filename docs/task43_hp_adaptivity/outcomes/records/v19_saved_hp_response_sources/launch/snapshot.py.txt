"""Write one immutable launch identity; no project/native imports or execution."""
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys

root, stage, contract = Path(sys.argv[1]), sys.argv[2], Path(sys.argv[3])
repo = Path('/home/fenics/Projects/NN-Lab-V2')
prefix = repo/'benchmarks/artifacts/task43/v19_saved_hp_response'
head = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for b in iter(lambda:f.read(1024*1024), b''):h.update(b)
    return h.hexdigest()
paths = [
    'benchmarks/__init__.py',
    'benchmarks/task43_hp_adaptivity/__init__.py',
    'src/__init__.py',
    'src/test/__init__.py',
    'src/io/__init__.py',
    'src/runners/__init__.py',
    'benchmarks/task43_hp_adaptivity/analyze_saved_hp_response.py',
    'src/test/test_task43_saved_hp_response.py',
    str(contract.relative_to(repo)),
    'scripts/activate_task40_phi90_pord64.sh',
    'benchmarks/subreaper_watchdog.py',
    'benchmarks/task034_wsl_resources.py',
    'benchmarks/task038_full3d_jit_staging.py',
    'src/runners/workflow_timebase.py',
    'src/io/native_capacity_profile.py',
]
snapshot = root/'snapshots';snapshot.mkdir()
refs = {}
for rel in paths:
    p=repo/rel
    if p.is_symlink():raise ValueError('symlink source '+rel)
    recorded=subprocess.check_output(['git','show',head+':'+rel])
    if p.read_bytes()!=recorded:raise ValueError('source differs from HEAD '+rel)
    out=snapshot/rel;out.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(p,out)
    refs[rel]={'sha256':sha(p),'bytes':p.stat().st_size}
launch_refs={}
for p in sorted((prefix/'launch').iterdir()):
    if p.suffix not in ('.sh','.py'):continue
    archived=repo/'docs/task43_hp_adaptivity/outcomes/records/v19_saved_hp_response_sources/launch'/(p.name+'.txt')
    archived_bytes=subprocess.check_output(['git','show',head+':'+str(archived.relative_to(repo))])
    if p.read_bytes()!=archived_bytes:raise ValueError('launch differs from committed archive '+p.name)
    out=snapshot/'launch'/p.name;out.parent.mkdir(exist_ok=True);shutil.copyfile(p,out)
    launch_refs[str(p)]={'sha256':sha(p),'bytes':p.stat().st_size}
names=['_MYFENICS_NATIVE_QUALIFIED_ACTIVATION','VIRTUAL_ENV','PATH','PYTHONPATH','LD_LIBRARY_PATH',
       'PHYSICAL_NATIVE_CAPACITY','PHYSICAL_NATIVE_NODE_CAP_BYTES','TASK43_V19_RUN_ROOT','TASK43_V19_STAGE',
       'OMP_NUM_THREADS','OPENBLAS_NUM_THREADS','MKL_NUM_THREADS','BLIS_NUM_THREADS','NUMEXPR_NUM_THREADS','VECLIB_MAXIMUM_THREADS']
d={'schema':'task43.v19.run-manifest.v1','stage':stage,'actual_head':head,
   'authorization_review':'5586112bf5bee924cc603253a5cbf7dd6c4a0863','sources':refs,'launch_files':launch_refs,
   'contract':{'path':str(contract),'sha256':sha(contract)},'worker_argv':sys.argv[5:],
   'worker_wrapper':sys.argv[4],'environment':{k:os.environ.get(k) for k in names},
   'worker_overrides':{'HWLOC_COMPONENTS':'-x86','OMPI_MCA_pml':'ob1','PYTEST_DISABLE_PLUGIN_AUTOLOAD':'1'},
   'mpi':'NOT_USED_NOT_ABI_QUALIFICATION','numpy_version':'RECORDED_IN_ACTUAL_WORKER_RUNTIME'}
p=root/'run_manifest.json'
with p.open('x') as f:json.dump(d,f,indent=2);f.write('\n')
