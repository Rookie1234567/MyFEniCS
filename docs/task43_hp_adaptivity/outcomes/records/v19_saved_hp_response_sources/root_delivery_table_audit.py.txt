dest = records / 'v19_saved_hp_response_sources'
base = repo / 'benchmarks/artifacts/task43/v19_saved_hp_response/support'

def read(p):
    return json.loads(p.read_text())

def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()

def write(p, obj):
    p.write_text(json.dumps(obj, ensure_ascii=False, indent=2, allow_nan=False) + '\n')

def close(x, y, rel=1e-10, absolute=1e-30):
    assert abs(x-y) <= max(absolute, rel*max(abs(x),abs(y))), (x,y)

stages = []
copied = []
for stage, name in [('T1','targeted_attempt1'),('D1','decomposition_attempt1'),('D2','decomposition_recovery1')]:
    root = base/name
    out = dest/name
    out.mkdir(exist_ok=True)
    # Existing copies must remain byte-identical; new evidence is copied verbatim.
    names = ['run_manifest.json','root_closeout.json','launcher_terminal.json',
             'runtime_start.json','runtime_end.json','watchdog/summary.json',
             'watchdog/workflow_phase.json','tests.xml','data/run_manifest.json']
    if stage == 'D2':
        names += ['data/diagnostic.json'] + [f'data/q{q}/{n}' for q in (8,10) for n in ('diagnostic.json','cells.csv','modes_612.csv')]
    for name2 in names + ['watchdog/worker.log']:
        src=root/name2
        if not src.exists():
            continue
        target=out/(name2+'.txt' if name2.endswith('.log') else name2)
        target.parent.mkdir(parents=True,exist_ok=True)
        if target.exists():
            assert target.read_bytes()==src.read_bytes(), target
        else:
            shutil.copyfile(src,target)
        copied.append({'source':str(src.relative_to(repo)), 'copy':str(target.relative_to(repo)), 'sha256':sha(src), 'bytes':src.stat().st_size})
    # Small complete new traces, including every stack and exit tail, are reviewable.
    for src in sorted((root/'strace').glob('sched.*')):
        target=out/(src.name+'.txt')
        if target.exists(): assert target.read_bytes()==src.read_bytes()
        else: shutil.copyfile(src,target)
        copied.append({'source':str(src.relative_to(repo)), 'copy':str(target.relative_to(repo)), 'sha256':sha(src),'bytes':src.stat().st_size})
    terminal=read(root/'launcher_terminal.json')
    summary=read(root/'watchdog/summary.json')
    audit=read(root/'root_closeout.json')
    manifest=read(root/'run_manifest.json')
    assert summary['descendants_cleared'] and not summary['remaining_child_pids']
    assert summary['sampled_process_tree_swap_peak_bytes']==0
    assert summary['sampled_process_tree_rss_peak_bytes'] <= 8*1024**3
    if stage != 'D1': assert not audit['issues'], audit['issues']
    stages.append({'stage':stage,'root':str(root.relative_to(repo)),
      'source_head':audit['source_head'], 'whole_seconds':terminal['entry_elapsed_seconds'],
      'whole_limit_seconds':terminal['whole_limit_seconds'], 'watchdog_seconds_nested':summary['elapsed_seconds'],
      'sampled_tree_rss_peak_bytes':summary['sampled_process_tree_rss_peak_bytes'],
      'sampled_tree_swap_peak_bytes':summary['sampled_process_tree_swap_peak_bytes'],
      'global_swap_activity':summary['global_swap_activity'],
      'exit_codes':{'worker':summary['leader_exit_code'],'watchdog':terminal['watchdog_exit_code'],'supervisor':terminal['supervisor_exit_code']},
      'received_signal':terminal['received_signal'], 'descendants_cleared':True,
      'root_audit':str((out/'root_closeout.json').relative_to(repo)),
      'raw_trace_lines_read':audit['trace_lines_read'], 'raw_resource_sha256':audit['resources_sha256'],
      'root_audit_issues_as_saved':audit['issues'],
      'D1_issue_classification':'Known checker owner-index rejection, not unknown exit; see d2_recovery_decision.json' if stage=='D1' else None,
      'tests':{'T1':'12 passed; initial frozen bytes','D1':'NOT_RUN','D2':'3 affected nodes passed in same worker before checker'}[stage]})

for q in (8,10):
    root=base/'decomposition_recovery1/data'/f'q{q}'
    d=read(root/'diagnostic.json')
    with (root/'cells.csv').open() as f: rows=list(csv.DictReader(f))
    with (root/'modes_612.csv').open() as f: modes=list(csv.DictReader(f))
    assert len(rows)==180 and {int(r['global_cell_id']) for r in rows}==set(range(180))
    assert len(modes)==612 and len({(r['side'],r['m'],r['n'],r['polarization']) for r in modes})==612

    for action in ('H8','P8'):
        sums[action]={}
        for comp in ('E','curl_over_k0','G'):
            fields=('B','D','X','I','after_error_squared','I_independent')
            values={k:math.fsum(float(r[f'{action}.{comp}.{k}']) for r in rows) for k in fields}
            maximum=0.0
            for r in rows:
                v={k:float(r[f'{action}.{comp}.{k}']) for k in fields}
                assert all(math.isfinite(x) for x in v.values())
                scale=max(v['B'],v['after_error_squared'],abs(v['X']),v['D'],1e-30)
                discrepancy=max(abs((v['X']-v['D'])-(v['B']-v['after_error_squared'])),abs(v['I']-(v['X']-v['D'])),abs(v['I_independent']-(v['B']-v['after_error_squared'])))
                maximum=max(maximum,discrepancy/scale)
                assert discrepancy<=1e-10*scale
            maxima[f'{action}.{comp}']=maximum
            stored=d['actions'][action]['G_identity_checks'][comp]
            for key in fields:
                close(values[key],stored['sum_'+key])
            for region in d['block_response_summaries']:
                subset=[r for r in rows if r['region']==region]
                block=d['block_response_summaries'][region][action][comp]
                assert len(subset)==block['cell_count']
                for key in fields:
                    close(math.fsum(float(r[f'{action}.{comp}.{key}']) for r in subset),block[key])
            sums[action][comp]=values
        G=math.sqrt(sums[action]['G']['after_error_squared']/d['cell_partition']['normalized_G_squared_denominator'])
        close(G,d['actions'][action]['G_eta_saved'],3e-11,1e-15)
        # E+curl must equal G per cell, not just their global aggregate.
        for r in rows:
            for k in fields:
                x=float(r[f'{action}.E.{k}'])+float(r[f'{action}.curl_over_k0.{k}'])
                y=float(r[f'{action}.G.{k}'])
                close(x,y)
    exceeds=[]
    max_amplitude_identity_error=0.0
    for r in modes:
        for a in ('R9','H8','P8'):
            ref=complex(float(r['R9_amplitude_real']),float(r['R9_amplitude_imag']))
            amp=complex(float(r[a+'_amplitude_real']),float(r[a+'_amplitude_imag']))
            for other in ('R9','Y6_1'):
                signed=float(r[a+'_saved_power_ratio'])-float(r[other+'_saved_power_ratio'])
                close(signed,float(r[f'{a}_minus_{other}_power_ratio_signed']))
                od=amp-complex(float(r[other+'_amplitude_real']),float(r[other+'_amplitude_imag']))
                close(od.real,float(r[f'{a}_minus_{other}_amplitude_real']))
                close(od.imag,float(r[f'{a}_minus_{other}_amplitude_imag']))
            delta=amp-ref
            direct=abs(amp)**2-abs(ref)**2
            expanded=2*(ref.conjugate()*delta).real+abs(delta)**2
            max_amplitude_identity_error=max(max_amplitude_identity_error,abs(direct-expanded))
            close(direct,expanded,3e-11,1e-15)
            diff=float(r[f'{a}_minus_Y6_1_power_ratio_signed'])
            if abs(diff)>float(r['power_threshold']):
                exceeds.append({'action':a,'side':r['side'],'m':int(r['m']),'n':int(r['n']),'polarization':r['polarization'],'signed_difference':diff,'absolute_difference':abs(diff),'threshold':float(r['power_threshold'])})
    key=lambda x:(x['action'],x['side'],x['m'],x['n'],x['polarization'])
    assert sorted(exceeds,key=key)==sorted(d['power_channels_over_1e-6'],key=key)
    for action in ('R9','H8','P8'):
        jr=d['actions'][action]['J_replay']
        assert len(jr['components'])==16
        close(max(jr['components'].values()),jr['value'])
    top=sorted(rows,key=lambda r:float(r['H8.G.B']),reverse=True)[:10]
    totalB=d['cell_partition']['baseline_R9_to_Y6_1_error_G_squared']
    z={}
    for r in rows:
        k=str((json.loads(r['lower_nm'])[2],json.loads(r['upper_nm'])[2]))
        z[k]=z.get(k,0)+float(r['H8.G.B'])/totalB
    audits[f'q{q}']={'cells':180,'channels':612,'identity_relative_limit':1e-10,
      'cell_identity_max_relative_error':maxima,'CSV_global_sums':sums,
      'all_power_exceedances_independently_enumerated':exceeds,
      'amplitude_squared_identity_max_absolute_error':max_amplitude_identity_error,
      'original_error_share_by_z_interval_nm':z,
      'top10_original_error_cells':[{'global_cell_id':int(r['global_cell_id']),'lower_nm':json.loads(r['lower_nm']),'B_share':float(r['H8.G.B'])/totalB} for r in top],
      'top10_original_error_share':math.fsum(float(r['H8.G.B']) for r in top)/totalB,
      'root_scope':'Recomputed completed CSV/JSON scalar arithmetic using stdlib; NPZ array checks performed by D2 worker, not reexecuted by root.'}
    science[f'q{q}']={k:d[k] for k in ('quadrature','cell_partition','block_response_summaries','actions','power_channels_over_1e-6')}
