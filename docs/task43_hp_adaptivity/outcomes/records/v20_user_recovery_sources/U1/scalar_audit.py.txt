def load(p):return json.loads(p.read_text())
def close(a,b,label):
 assert math.isfinite(a) and math.isfinite(b)
 assert abs(a-b)<=max(1e-15,3e-11*abs(b)),(label,a,b)
 return {'actual':a,'expected':b,'absolute_error':abs(a-b)}

av={n:load(repo/c['sources'][role+'_absorption']['path'])['A_volume_total'] for n,role in [('R9','r9'),('R10','r10')]};av['Y6_1']=load(repo/c['sources']['y6_summary']['path'])['official_result']['volume_metrics']['A_volume_total']

output={}
for q in ['q8','q10']:
 d=load(r/'data'/q/'diagnostic.json');cells=list(csv.DictReader((r/'data'/q/'cells.csv').open()));rows=list(csv.DictReader((r/'data'/q/'modes_612.csv').open()));qs={}
 for n in ['R9','R10']:
  a=c['anchors'][n];components={};checks={}
  for name,scale in tau.items():
   value=av[n]-av['Y6_1'] if name=='A_volume' else modal[n]['metrics'][name]-modal['Y6_1']['metrics'][name]
   components['power.'+name]=abs(value)/scale
  components['power.max_individual_channel']=max(abs(float(row[n+'_power_ratio'])-float(row['Y6_1_power_ratio'])) for row in rows)/1e-6
  for side in ['top','bottom']:
   for pol in ['s','p']:
    v=[(float(row[n+'_amplitude_real'])-float(row['Y6_1_amplitude_real']))**2+(float(row[n+'_amplitude_imag'])-float(row['Y6_1_amplitude_imag']))**2 for row in rows if row['side']==side and row['polarization']==pol]
    assert len(v)==153;components['amplitude.'+side+'_'+pol]=math.sqrt(math.fsum(v))/1e-4
  e=math.sqrt(math.fsum(float(row[n+'.E.B10']) for row in cells));h=c['physics']['magnetic_field_scale_a_per_m']*math.sqrt(math.fsum(float(row[n+'.curl_over_k0.B10']) for row in cells));g=math.sqrt(math.fsum(float(row[n+'.G.B10']) for row in cells))
  for name,norm in [('E_total_code',e),('E_scattered_code',e),('H_total_A_per_m',h),('H_scattered_A_per_m',h)]:
   anchor=a['field_metrics'][name];ratio=norm/max(anchor['reference_l2'],anchor['reference_norm_floor']);components['field.'+name]=ratio/1e-4
   checks[name]=close(ratio,d['field_and_J_replay'][n]['field_relative_metrics'][name],q+n+name)
  ga=a['field_metrics']['G'];components['field.G']=g/max(ga['reference_g_norm'],ga['incident_g_norm_floor'])/1e-4
  assert len(components)==16
  for name,value in components.items():
   checks[name+'.current']=close(value,d['field_and_J_replay'][n]['J_replay']['components'][name],q+n+name+'current')
   checks[name+'.old']=close(value,a['J_infinity']['components'][name],q+n+name+'old')
  witness=max(components,key=components.__getitem__);assert witness==a['J_infinity']['witness'];j=max(components.values());close(j,d['field_and_J_replay'][n]['J_replay']['value'],q+n+'J')
  qs[n]={'J':j,'witness':witness,'components':components,'checks':checks}
 layers={}
 for name,lr in d['z_layer_summaries'].items():
  z0,z1=lr['z_interval_nm'];subset=[row for row in cells if json.loads(row['lower_nm'])[2]==z0 and json.loads(row['upper_nm'])[2]==z1];assert len(subset)==lr['cell_count'];terms={}
  for term in ['B','D','X','I','B10','I_independent']:

  impacts=[float(row['R10.G.I']) for row in subset];terms['positive_I']=math.fsum(v for v in impacts if v>0);terms['negative_I']=math.fsum(v for v in impacts if v<0)

 for name,g in d['degree_summaries'].items():
