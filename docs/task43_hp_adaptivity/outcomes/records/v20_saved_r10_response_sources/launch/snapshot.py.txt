 with p.open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
c=json.loads(contract.read_text())
assert c['schema']=='task43.v20.saved_r10_response_contract.v1'

 out=snapshot/rel;out.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(p,out)
